export function respondida(r: Respuesta | undefined): boolean {
  if (!r) return false;
  switch (r.tipo) {
    case 'opcion':
      return !!r.id;
    case 'opciones':
    case 'orden':
      return r.ids.length > 0;
    case 'booleano':
      return r.valor !== null;
    case 'texto':
      return r.valor.trim() !== '';
    case 'pares':
      return Object.keys(r.asignacion).length > 0;
    case 'caso':
      return Object.values(r.respuestas).some(respondida);
  }
}
